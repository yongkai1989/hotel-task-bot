-- Front Office Schedule
-- Run once in Supabase SQL Editor. Safe to run again.

alter table public.user_profiles
  add column if not exists can_access_fo_schedule boolean not null default false,
  add column if not exists fo_schedule_view_only boolean not null default false;

create table if not exists public.fo_schedule_staff (
  id uuid primary key default gen_random_uuid(),
  staff_name text not null,
  staff_role text not null default 'FRONT_OFFICE'
    check (staff_role in ('MANAGER', 'FRONT_OFFICE', 'NIGHT_AUDITOR', 'RESERVATIONS')),
  staff_gender text constraint fo_schedule_staff_gender_check check (staff_gender in ('MALE', 'FEMALE')),
  fixed_off_day integer check (fixed_off_day between 1 and 7),
  preferred_shift text not null default 'FLEX'
    check (preferred_shift in ('FLEX', 'MORNING', 'AFTERNOON', 'NIGHT', 'RESERVATIONS')),
  is_permanent_night boolean not null default false,
  is_hybrid_night boolean not null default false,
  is_manager boolean not null default false,
  is_active boolean not null default true,
  sort_order integer not null default 0,
  created_by_user_id uuid not null,
  created_by_name text not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table public.fo_schedule_staff
  add column if not exists staff_gender text;

      and conname = 'fo_schedule_staff_gender_check'

    alter table public.fo_schedule_staff
      add constraint fo_schedule_staff_gender_check
      check (staff_gender in ('MALE', 'FEMALE'));
  end if;
end $$;

-- More than one staff member may be eligible to rotate into the hybrid night role.
-- Dropping this legacy unique index does not change any staff details or schedules.
drop index if exists public.fo_schedule_one_hybrid_night_idx;
create index if not exists fo_schedule_staff_active_order_idx
  on public.fo_schedule_staff (is_active, is_manager, sort_order, staff_name);

create table if not exists public.fo_schedule_shifts (
  id uuid primary key default gen_random_uuid(),
  shift_name text not null,
  shift_code text not null,
  start_time time not null,
  end_time time not null,
  color text not null default 'BLUE'
    check (color in ('BLUE', 'TEAL', 'PURPLE', 'AMBER', 'PINK', 'SLATE')),
  is_active boolean not null default true,
  created_by_user_id uuid not null,
  created_by_name text not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
create unique index if not exists fo_schedule_shift_name_unique
  on public.fo_schedule_shifts (lower(shift_name));
create unique index if not exists fo_schedule_shift_code_unique
  on public.fo_schedule_shifts (upper(shift_code));

create table if not exists public.fo_schedule_entries (
  id uuid primary key default gen_random_uuid(),
  staff_id uuid not null references public.fo_schedule_staff(id) on delete restrict,
  schedule_date date not null,
  status text not null check (status in ('WORK', 'AL', 'UPL', 'NO_SHOW', 'MC', 'OFF')),
  shift_id uuid references public.fo_schedule_shifts(id) on delete set null,
  shift_name_snapshot text,
  shift_code_snapshot text,
  scheduled_start time,
  scheduled_end time,
  is_late boolean not null default false,
  overtime_minutes integer not null default 0 check (overtime_minutes between 0 and 1440),
  notes text,
  auto_filled boolean not null default false,
  compliance_exception boolean not null default false,
  updated_by_user_id uuid not null,
  updated_by_name text not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (staff_id, schedule_date)
);
create index if not exists fo_schedule_entries_date_staff_idx
  on public.fo_schedule_entries (schedule_date, staff_id);
create index if not exists fo_schedule_entries_staff_date_idx
  on public.fo_schedule_entries (staff_id, schedule_date desc);

create or replace function public.fo_schedule_actor_name()
returns text language sql stable security definer set search_path = public
as $$
  select coalesce(nullif(trim(up.name), ''), nullif(trim(up.email), ''), 'User')
  from public.user_profiles up where up.user_id = (select auth.uid()) limit 1
$$;

create or replace function public.can_access_fo_schedule()
returns boolean language sql stable security definer set search_path = public
as $$
  select exists (
    select 1 from public.user_profiles up
    where up.user_id = (select auth.uid())
      and (upper(coalesce(up.role, '')) = 'SUPERUSER' or up.can_access_fo_schedule)
  )
$$;

create or replace function public.can_manage_fo_schedule()
returns boolean language sql stable security definer set search_path = public
as $$
  select exists (
    select 1 from public.user_profiles up
    where up.user_id = (select auth.uid())
      and (
        upper(coalesce(up.role, '')) = 'SUPERUSER'
        or (up.can_access_fo_schedule and not up.fo_schedule_view_only)
      )
  )
$$;

create or replace function public.save_fo_schedule_staff(
  p_staff_id uuid, p_staff_name text, p_staff_role text,
  p_fixed_off_day integer, p_preferred_shift text,
  p_is_permanent_night boolean default false,
  p_is_hybrid_night boolean default false,
  p_is_manager boolean default false,
  p_is_active boolean default true,
  p_staff_gender text default null
) returns public.fo_schedule_staff
language plpgsql security definer set search_path = public
as $$
declare
  v_row public.fo_schedule_staff;
  v_old_fixed_off_day integer;
  v_name text := trim(coalesce(p_staff_name, ''));
  v_role text := upper(trim(coalesce(p_staff_role, 'FRONT_OFFICE')));
  v_preference text := upper(trim(coalesce(p_preferred_shift, 'FLEX')));
  v_gender text := nullif(upper(trim(coalesce(p_staff_gender, ''))), '');
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  if v_name = '' then raise exception 'Staff name is required'; end if;
  if v_role not in ('MANAGER','FRONT_OFFICE','NIGHT_AUDITOR','RESERVATIONS') then raise exception 'Invalid Front Office role'; end if;
  if v_preference not in ('FLEX','MORNING','AFTERNOON','NIGHT','RESERVATIONS') then raise exception 'Invalid preferred shift'; end if;
  if v_gender is not null and v_gender not in ('MALE','FEMALE') then raise exception 'Gender must be Male or Female'; end if;
  if p_fixed_off_day is not null and p_fixed_off_day not between 1 and 7 then raise exception 'Invalid fixed off day'; end if;
  if p_fixed_off_day=6 then raise exception 'Saturday cannot be selected as a fixed off day'; end if;
  if coalesce(p_is_permanent_night, false) then v_preference := 'NIGHT'; v_role := 'NIGHT_AUDITOR'; end if;
  if coalesce(p_is_manager, false) then v_role := 'MANAGER'; end if;

  if p_staff_id is null then
    insert into public.fo_schedule_staff (
      staff_name, staff_role, staff_gender, fixed_off_day, preferred_shift,
      is_permanent_night, is_hybrid_night, is_manager, is_active,
      sort_order, created_by_user_id, created_by_name
    ) values (
      v_name, v_role, v_gender, p_fixed_off_day, v_preference,
      coalesce(p_is_permanent_night,false), coalesce(p_is_hybrid_night,false),
      coalesce(p_is_manager,false), coalesce(p_is_active,true),
      coalesce((select max(sort_order)+1 from public.fo_schedule_staff),1),
      (select auth.uid()), public.fo_schedule_actor_name()
    ) returning * into v_row;
  else
    select fixed_off_day into v_old_fixed_off_day
    from public.fo_schedule_staff
    where id=p_staff_id;

    update public.fo_schedule_staff set
      staff_name=v_name, staff_role=v_role, staff_gender=coalesce(v_gender,staff_gender), fixed_off_day=p_fixed_off_day,
      preferred_shift=v_preference, is_permanent_night=coalesce(p_is_permanent_night,false),
      is_hybrid_night=coalesce(p_is_hybrid_night,false), is_manager=coalesce(p_is_manager,false),
      is_active=coalesce(p_is_active,true), updated_at=now()
    where id=p_staff_id returning * into v_row;
    if not found then raise exception 'Staff record not found'; end if;

    -- Fixed off is authoritative for Auto Fill. Reclassify only the affected
    -- old/new weekday cells as generated cells so the next Auto Fill can
    -- rebuild them. Manual leave, MC, UPL and No Show records are preserved.
    if v_old_fixed_off_day is distinct from p_fixed_off_day then
      update public.fo_schedule_entries e
      set auto_filled=true, updated_by_user_id=(select auth.uid()),
          updated_by_name=public.fo_schedule_actor_name(), updated_at=now()
      where e.staff_id=p_staff_id
        and e.schedule_date >= date_trunc('month',current_date)::date
        and e.status='OFF'
        and v_old_fixed_off_day is not null
        and extract(isodow from e.schedule_date)::integer=v_old_fixed_off_day;

      update public.fo_schedule_entries e
      set auto_filled=true, updated_by_user_id=(select auth.uid()),
          updated_by_name=public.fo_schedule_actor_name(), updated_at=now()
      where e.staff_id=p_staff_id
        and e.schedule_date >= date_trunc('month',current_date)::date
        and e.status in ('WORK','OFF')
        and p_fixed_off_day is not null
        and extract(isodow from e.schedule_date)::integer=p_fixed_off_day;
    end if;
  end if;
  return v_row;
end $$;

create or replace function public.delete_fo_schedule_staff(p_staff_id uuid)
returns text language plpgsql security definer set search_path=public as $$
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  if exists (select 1 from public.fo_schedule_entries where staff_id=p_staff_id) then
    update public.fo_schedule_staff set is_active=false, is_hybrid_night=false, updated_at=now() where id=p_staff_id;
    return 'ARCHIVED';
  end if;
  delete from public.fo_schedule_staff where id=p_staff_id;
  return 'DELETED';
end $$;

create or replace function public.save_fo_schedule_shift(
  p_shift_id uuid, p_shift_name text, p_shift_code text,
  p_start_time time, p_end_time time, p_color text, p_is_active boolean default true
) returns public.fo_schedule_shifts language plpgsql security definer set search_path=public as $$
declare v_row public.fo_schedule_shifts; v_color text:=upper(coalesce(p_color,'BLUE'));
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  if trim(coalesce(p_shift_name,''))='' or trim(coalesce(p_shift_code,''))='' then raise exception 'Shift name and code are required'; end if;
  if v_color not in ('BLUE','TEAL','PURPLE','AMBER','PINK','SLATE') then raise exception 'Invalid colour'; end if;
  if p_shift_id is null then
    insert into public.fo_schedule_shifts(shift_name,shift_code,start_time,end_time,color,is_active,created_by_user_id,created_by_name)
    values(trim(p_shift_name),upper(trim(p_shift_code)),p_start_time,p_end_time,v_color,coalesce(p_is_active,true),(select auth.uid()),public.fo_schedule_actor_name())
    returning * into v_row;
  else
    update public.fo_schedule_shifts set shift_name=trim(p_shift_name),shift_code=upper(trim(p_shift_code)),start_time=p_start_time,end_time=p_end_time,color=v_color,is_active=coalesce(p_is_active,true),updated_at=now()
    where id=p_shift_id returning * into v_row;
    if not found then raise exception 'Shift not found'; end if;
  end if;
  return v_row;
end $$;

create or replace function public.delete_fo_schedule_shift(p_shift_id uuid)
returns text language plpgsql security definer set search_path=public as $$
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  if exists(select 1 from public.fo_schedule_entries where shift_id=p_shift_id) then
    update public.fo_schedule_shifts set is_active=false,updated_at=now() where id=p_shift_id; return 'ARCHIVED';
  end if;
  delete from public.fo_schedule_shifts where id=p_shift_id; return 'DELETED';
end $$;

create or replace function public.save_fo_schedule_entry(
  p_staff_id uuid, p_schedule_date date, p_status text,
  p_shift_id uuid default null, p_overtime_minutes integer default 0,
  p_notes text default null, p_custom_start time default null,
  p_custom_end time default null, p_is_late boolean default false
) returns public.fo_schedule_entries language plpgsql security definer set search_path=public as $$
declare
  v_row public.fo_schedule_entries; v_shift public.fo_schedule_shifts;
  v_status text:=upper(trim(coalesce(p_status,''))); v_name text; v_code text; v_start time; v_end time;
  v_staff_name text;
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  select staff_name into v_staff_name from public.fo_schedule_staff where id=p_staff_id;
  if not found then raise exception 'Staff not found'; end if;
  if v_status not in ('WORK','AL','UPL','NO_SHOW','MC','OFF') then raise exception 'Invalid status'; end if;
  if v_status='WORK' then
    if p_shift_id is not null then select * into v_shift from public.fo_schedule_shifts where id=p_shift_id;
      if not found then raise exception 'Choose a valid shift'; end if;
      v_name:=v_shift.shift_name; v_code:=v_shift.shift_code; v_start:=v_shift.start_time; v_end:=v_shift.end_time;
    else
      if p_custom_start is null or p_custom_end is null then raise exception 'Custom hours are required'; end if;
      v_name:='Ad hoc hours'; v_code:=to_char(p_custom_start,'HH24:MI')||'-'||to_char(p_custom_end,'HH24:MI'); v_start:=p_custom_start; v_end:=p_custom_end;
    end if;

    -- This pairing is prohibited even for manual edits. The Auto Fill function
    -- applies the same rule, while the UI continues to flag incomplete Night pairs.
    if upper(coalesce(v_code,''))='NIGHT'
      and lower(trim(v_staff_name)) in ('jia','saravanan')
      and exists(
        select 1
        from public.fo_schedule_entries e
        join public.fo_schedule_staff other_staff on other_staff.id=e.staff_id
        where e.schedule_date=p_schedule_date
          and e.staff_id<>p_staff_id
          and e.status='WORK'
          and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
          and lower(trim(other_staff.staff_name))=
            case when lower(trim(v_staff_name))='jia' then 'saravanan' else 'jia' end
      ) then
      raise exception 'Jia and Saravanan cannot be assigned to the same Night shift';
    end if;
  else p_shift_id:=null; p_is_late:=false; p_overtime_minutes:=0; end if;
  insert into public.fo_schedule_entries(staff_id,schedule_date,status,shift_id,shift_name_snapshot,shift_code_snapshot,scheduled_start,scheduled_end,is_late,overtime_minutes,notes,auto_filled,compliance_exception,updated_by_user_id,updated_by_name)
  values(p_staff_id,p_schedule_date,v_status,p_shift_id,case when v_status='WORK' then v_name end,case when v_status='WORK' then v_code end,case when v_status='WORK' then v_start end,case when v_status='WORK' then v_end end,coalesce(p_is_late,false),greatest(0,least(coalesce(p_overtime_minutes,0),1440)),nullif(trim(coalesce(p_notes,'')),''),false,false,(select auth.uid()),public.fo_schedule_actor_name())
  on conflict(staff_id,schedule_date) do update set status=excluded.status,shift_id=excluded.shift_id,shift_name_snapshot=excluded.shift_name_snapshot,shift_code_snapshot=excluded.shift_code_snapshot,scheduled_start=excluded.scheduled_start,scheduled_end=excluded.scheduled_end,is_late=excluded.is_late,overtime_minutes=excluded.overtime_minutes,notes=excluded.notes,auto_filled=false,compliance_exception=false,updated_by_user_id=excluded.updated_by_user_id,updated_by_name=excluded.updated_by_name,updated_at=now()
  returning * into v_row; return v_row;
end $$;

create or replace function public.delete_fo_schedule_entry(p_staff_id uuid,p_schedule_date date)
returns boolean language plpgsql security definer set search_path=public as $$
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  delete from public.fo_schedule_entries where staff_id=p_staff_id and schedule_date=p_schedule_date; return found;
end $$;

create or replace function public.fill_fo_schedule_range(
  p_staff_id uuid,p_start_date date,p_end_date date,p_weekdays integer[],
  p_status text,p_shift_id uuid default null,p_apply_fixed_off boolean default true
) returns integer language plpgsql security definer set search_path=public as $$
declare v_date date; v_count integer:=0; v_fixed integer;
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  if p_end_date<p_start_date or p_end_date>p_start_date+62 then raise exception 'Range must be 1 to 63 days'; end if;
  select fixed_off_day into v_fixed from public.fo_schedule_staff where id=p_staff_id;
  if not found then raise exception 'Staff not found'; end if;
  for v_date in select generate_series(p_start_date,p_end_date,interval '1 day')::date loop
    if coalesce(p_apply_fixed_off,true) and v_fixed is not null and extract(isodow from v_date)::integer=v_fixed then
      perform public.save_fo_schedule_entry(p_staff_id,v_date,'OFF',null,0,null); v_count:=v_count+1;
    elsif p_weekdays is null or cardinality(p_weekdays)=0 or extract(isodow from v_date)::integer=any(p_weekdays) then
      perform public.save_fo_schedule_entry(p_staff_id,v_date,p_status,p_shift_id,0,null); v_count:=v_count+1;
    end if;
  end loop; return v_count;
end $$;

create or replace function public.cleanup_fo_schedule_history()
returns integer language plpgsql security definer set search_path=public as $$
declare v_count integer;
begin
  if not public.can_manage_fo_schedule() then return 0; end if;
  delete from public.fo_schedule_entries where schedule_date < (current_date - interval '6 months')::date;
  get diagnostics v_count=row_count; return v_count;
end $$;

create or replace function public.fo_schedule_rest_hours(
  p_previous_date date,
  p_previous_start time,
  p_previous_end time,
  p_current_date date,
  p_current_start time
) returns numeric
language sql immutable security invoker set search_path=public
as $$
  select case
    when p_previous_date is null or p_previous_start is null or p_previous_end is null
      or p_current_date is null or p_current_start is null then null
    else round(extract(epoch from (
      (p_current_date::timestamp + p_current_start)
      - (p_previous_date::timestamp + p_previous_end
         + case when p_previous_end<=p_previous_start then interval '1 day' else interval '0 days' end)
    )) / 3600.0, 2)
  end
$$;

create or replace function public.autofill_fo_schedule_month(p_month date)
returns jsonb language plpgsql security definer set search_path=public as $$
declare
  v_start date:=date_trunc('month',p_month)::date;
  v_end date:=(date_trunc('month',p_month)+interval '1 month - 1 day')::date;
  v_date date; v_shift_code text; v_target integer; v_current integer; v_staff_id uuid;
  v_shift public.fo_schedule_shifts; v_inserted integer:=0; v_exceptions integer:=0; v_shortages integer:=0;
  v_prev_code text; v_prev_date date; v_prev_start time; v_prev_end time;
  v_rest_hours numeric; v_is_exception boolean; v_night_male integer:=0; v_missing_gender text;
  v_replaced integer:=0; v_core_ready boolean:=false;
  v_morning_count integer:=0; v_afternoon_count integer:=0;
  v_night_count integer:=0; v_reservations_count integer:=0;
  v_manager_target text; v_rebalance_target text; v_rebalance_entry_id uuid;
  v_saravanan_solo boolean:=false; v_saravanan_present boolean:=false;
  v_cover_mode boolean:=false; v_solo_nights_week integer:=0;
begin
  if not public.can_manage_fo_schedule() then raise exception 'You do not have permission to manage Front Office Schedule'; end if;
  if not exists(select 1 from public.fo_schedule_staff where is_active and is_permanent_night) then raise exception 'Set a permanent night auditor before Auto Fill'; end if;
  if not exists(select 1 from public.fo_schedule_staff where is_active and is_hybrid_night) then raise exception 'Choose at least one hybrid night staff member before Auto Fill'; end if;
  select string_agg(staff_name, ', ' order by sort_order, staff_name) into v_missing_gender
  from public.fo_schedule_staff where is_active and staff_gender is null;
  if v_missing_gender is not null then
    raise exception 'Set Male or Female for every active staff member before Auto Fill: %', v_missing_gender;
  end if;
  foreach v_shift_code in array array['MORNING','AFTERNOON','NIGHT','RESERVATIONS','MID','MANAGER','COVER_MORNING','COVER_NOON','COVER_MANAGER'] loop
    if not exists(select 1 from public.fo_schedule_shifts where is_active and upper(shift_code)=v_shift_code) then raise exception 'Required shift % is missing',v_shift_code; end if;
  end loop;

  -- Rebuild only previous Auto Fill output. Every manually saved cell has
  -- auto_filled=false and remains exactly as entered by the manager.
  delete from public.fo_schedule_entries
  where schedule_date between v_start and v_end and auto_filled=true;
  get diagnostics v_replaced=row_count;

  for v_date in select generate_series(v_start,v_end,interval '1 day')::date loop
    insert into public.fo_schedule_entries(staff_id,schedule_date,status,auto_filled,updated_by_user_id,updated_by_name)
    select s.id,v_date,'OFF',true,(select auth.uid()),public.fo_schedule_actor_name()
    from public.fo_schedule_staff s
    where s.is_active and s.fixed_off_day=extract(isodow from v_date)::integer
      and not exists(select 1 from public.fo_schedule_entries e where e.staff_id=s.id and e.schedule_date=v_date)
    on conflict do nothing;
    get diagnostics v_current=row_count; v_inserted:=v_inserted+v_current;

    select * into v_shift from public.fo_schedule_shifts where is_active and upper(shift_code)='NIGHT' limit 1;
    insert into public.fo_schedule_entries(staff_id,schedule_date,status,shift_id,shift_name_snapshot,shift_code_snapshot,scheduled_start,scheduled_end,auto_filled,updated_by_user_id,updated_by_name)
    select s.id,v_date,'WORK',v_shift.id,v_shift.shift_name,v_shift.shift_code,v_shift.start_time,v_shift.end_time,true,(select auth.uid()),public.fo_schedule_actor_name()
    from public.fo_schedule_staff s
    where s.is_active and s.is_permanent_night
      and not exists(select 1 from public.fo_schedule_entries e where e.staff_id=s.id and e.schedule_date=v_date)
      and not exists(
        select 1
        from public.fo_schedule_entries e
        join public.fo_schedule_staff j on j.id=e.staff_id
        where e.schedule_date=v_date and e.status='WORK'
          and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
          and lower(trim(j.staff_name))='jia'
      )
    on conflict do nothing;
    get diagnostics v_current=row_count; v_inserted:=v_inserted+v_current;

    foreach v_shift_code in array array['NIGHT','RESERVATIONS','MORNING','AFTERNOON'] loop
      select * into v_shift
      from public.fo_schedule_shifts
      where is_active and upper(shift_code)=v_shift_code
      limit 1;
      v_target:=case v_shift_code when 'RESERVATIONS' then 1 else 2 end;
      select count(*) into v_current from public.fo_schedule_entries e
      where e.schedule_date=v_date and e.status='WORK' and upper(coalesce(e.shift_code_snapshot,''))=v_shift_code;
      if v_shift_code='NIGHT' then
        select count(*) into v_night_male
        from public.fo_schedule_entries e
        join public.fo_schedule_staff s on s.id=e.staff_id
        where e.schedule_date=v_date and e.status='WORK'
          and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
          and s.staff_gender='MALE';
      else
        v_night_male:=0;
      end if;
      while v_current<v_target loop
        v_is_exception:=false;
        select s.id into v_staff_id
        from public.fo_schedule_staff s
        left join lateral (
          select prior.*
          from public.fo_schedule_entries prior
          where prior.staff_id=s.id and prior.schedule_date<v_date and prior.status='WORK'
          order by prior.schedule_date desc
          limit 1
        ) prev on true
        where s.is_active and not s.is_permanent_night and not s.is_manager
          and not exists(select 1 from public.fo_schedule_entries e where e.staff_id=s.id and e.schedule_date=v_date)
          and (v_shift_code<>'NIGHT' or v_night_male>0 or s.staff_gender='MALE')
          and (
            v_shift_code<>'NIGHT'
            or not exists(
              select 1 from public.fo_schedule_entries e
              join public.fo_schedule_staff j on j.id=e.staff_id
              where e.schedule_date=v_date and e.status='WORK'
                and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
                and lower(trim(j.staff_name))='jia'
            )
            or lower(trim(s.staff_name)) in ('harish','zaim')
          )
          and (
            v_shift_code<>'NIGHT'
            or lower(trim(s.staff_name))<>'jia'
            or (
              not exists(
                select 1 from public.fo_schedule_entries e
                join public.fo_schedule_staff n on n.id=e.staff_id
                where e.schedule_date=v_date and e.status='WORK'
                  and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
                  and lower(trim(n.staff_name))='saravanan'
              )
              and exists(
                select 1 from public.fo_schedule_entries e
                join public.fo_schedule_staff p on p.id=e.staff_id
                where e.schedule_date=v_date and e.status='WORK'
                  and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
                  and lower(trim(p.staff_name)) in ('harish','zaim')
              )
            )
          )
          and not (
            prev.schedule_date is not null
            and upper(coalesce(prev.shift_code_snapshot,''))<>v_shift_code
            and public.fo_schedule_rest_hours(
              prev.schedule_date,prev.scheduled_start,prev.scheduled_end,v_date,v_shift.start_time
            )<16
          )
          and not (v_shift_code='MORNING' and lower(trim(s.staff_name))='sri')
          -- Reservations follows a strict staffing order. Jia is primary and
          -- Harish is secondary; Walter is handled separately below only when
          -- neither of them can cover the date.
          and (v_shift_code<>'RESERVATIONS' or lower(trim(s.staff_name)) in ('jia','harish'))
          -- Reserve Harish for Jia's Night pairing, Reservations cover, and
          -- the first available Mid assignment. Use him for Morning/Afternoon
          -- only through the last-resort coverage fallback.
          and not (v_shift_code in ('MORNING','AFTERNOON') and lower(trim(s.staff_name))='harish')
          -- Walter stays on Manager hours. He may cover Morning only through
          -- the final fallback; he is never automatically placed Afternoon.
          and not (v_shift_code in ('MORNING','AFTERNOON') and s.is_manager)
        order by s.is_manager,
          case
            -- If Jia was manually placed on Night, only an approved male
            -- partner can be selected and Zaim is preferred over Harish.
            when v_shift_code='NIGHT'
              and exists(
                select 1 from public.fo_schedule_entries e
                join public.fo_schedule_staff j on j.id=e.staff_id
                where e.schedule_date=v_date and e.status='WORK'
                  and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
                  and lower(trim(j.staff_name))='jia'
              )
              and lower(trim(s.staff_name))='zaim' then 0
            when v_shift_code='NIGHT'
              and exists(
                select 1 from public.fo_schedule_entries e
                join public.fo_schedule_staff j on j.id=e.staff_id
                where e.schedule_date=v_date and e.status='WORK'
                  and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
                  and lower(trim(j.staff_name))='jia'
              )
              and lower(trim(s.staff_name))='harish' then 1
            -- Normal Night rotation favours Sri, Zaim and Azalina. Harish is
            -- deliberately held back for Reservations cover or spare Mid duty.
            when v_shift_code='NIGHT' and lower(trim(s.staff_name)) in ('sri','zaim','azalina') then 0
            when v_shift_code='NIGHT' and s.is_hybrid_night and lower(trim(s.staff_name))<>'jia' then 1
            when v_shift_code='NIGHT' and lower(trim(s.staff_name))='jia' then 3
            when v_shift_code='NIGHT' and lower(trim(s.staff_name))='harish' then 4
            when v_shift_code='NIGHT' then 2
            -- Jia is the primary Reservations staff. Harish covers
            -- Reservations when Jia is off, on leave, or already assigned.
            when v_shift_code='RESERVATIONS' and lower(trim(s.staff_name))='jia' then 0
            when v_shift_code='RESERVATIONS' and lower(trim(s.staff_name))='harish' then 1
            when v_shift_code='RESERVATIONS' then 2
            -- Once Night and Reservations have been covered, Sri is preferred
            -- for Afternoon. Keep Harish available for Mid when manpower allows.
            when v_shift_code='AFTERNOON' and lower(trim(s.staff_name))='sri' then 0
            when v_shift_code in ('MORNING','AFTERNOON') and lower(trim(s.staff_name))='harish' then 3
            else 0
          end,
          case when s.preferred_shift=v_shift_code then 0 when s.preferred_shift='FLEX' then 1 else 2 end,
          case when upper(coalesce(prev.shift_code_snapshot,''))=v_shift_code then 0 else 1 end,
          case when v_shift_code='NIGHT' then (
            select count(*)
            from public.fo_schedule_entries month_night
            where month_night.staff_id=s.id
              and month_night.schedule_date between v_start and v_date-1
              and month_night.status='WORK'
              and upper(coalesce(month_night.shift_code_snapshot,''))='NIGHT'
          ) else 0 end,
          s.sort_order,s.staff_name limit 1;

        if v_staff_id is null and v_shift_code in ('MORNING','AFTERNOON') then
          select s.id into v_staff_id from public.fo_schedule_staff s
          left join lateral (
            select prior.*
            from public.fo_schedule_entries prior
            where prior.staff_id=s.id and prior.schedule_date<v_date and prior.status='WORK'
            order by prior.schedule_date desc
            limit 1
          ) prev on true
          where s.is_active and not s.is_permanent_night and not s.is_manager
            and not exists(select 1 from public.fo_schedule_entries e where e.staff_id=s.id and e.schedule_date=v_date)
          order by
            case
              when v_shift_code='MORNING' and lower(trim(s.staff_name))='sri' then 2
              when upper(coalesce(prev.shift_code_snapshot,''))<>v_shift_code
                and public.fo_schedule_rest_hours(
                  prev.schedule_date,prev.scheduled_start,prev.scheduled_end,v_date,v_shift.start_time
                )<16 then 1
              else 0
            end,
            case when lower(trim(s.staff_name))='harish' then 1 else 0 end,
            s.sort_order,s.staff_name limit 1;
        end if;
        exit when v_staff_id is null;
        select * into v_shift from public.fo_schedule_shifts where is_active and upper(shift_code)=v_shift_code limit 1;
        v_prev_code:=null; v_prev_date:=null; v_prev_start:=null; v_prev_end:=null; v_rest_hours:=null;
        select upper(coalesce(shift_code_snapshot,'')),schedule_date,scheduled_start,scheduled_end
          into v_prev_code,v_prev_date,v_prev_start,v_prev_end
        from public.fo_schedule_entries
        where staff_id=v_staff_id and schedule_date<v_date and status='WORK'
        order by schedule_date desc limit 1;
        v_rest_hours:=public.fo_schedule_rest_hours(v_prev_date,v_prev_start,v_prev_end,v_date,v_shift.start_time);
        v_is_exception:=coalesce(v_prev_code<>v_shift_code and v_rest_hours<16,false);
        insert into public.fo_schedule_entries(staff_id,schedule_date,status,shift_id,shift_name_snapshot,shift_code_snapshot,scheduled_start,scheduled_end,notes,auto_filled,compliance_exception,updated_by_user_id,updated_by_name)
        values(v_staff_id,v_date,'WORK',v_shift.id,v_shift.shift_name,v_shift.shift_code,v_shift.start_time,v_shift.end_time,
          case when v_is_exception then 'Rest exception: '||v_prev_code||' to '||v_shift_code||' provides only '||trim(to_char(v_rest_hours,'FM999990.00'))||' hours before the next shift.' end,
          true,v_is_exception,(select auth.uid()),public.fo_schedule_actor_name());
        if v_shift_code='NIGHT' and exists(select 1 from public.fo_schedule_staff where id=v_staff_id and staff_gender='MALE') then v_night_male:=v_night_male+1; end if;
        v_inserted:=v_inserted+1; if v_is_exception then v_exceptions:=v_exceptions+1; end if; v_current:=v_current+1; v_staff_id:=null;
      end loop;
    end loop;

    -- Overtime bridge shifts are used only after ordinary staffing cannot meet
    -- the core targets and only while Saravanan is working. Cover Morning spans
    -- Morning and Afternoon; Cover Noon spans late Afternoon and Night.
    select
      count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,'')) in ('MORNING','COVER_MORNING')),
      count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,'')) in ('AFTERNOON','COVER_MORNING','COVER_NOON','COVER_MANAGER')),
      count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,'')) in ('NIGHT','COVER_NOON')),
      count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,''))='RESERVATIONS')
    into v_morning_count,v_afternoon_count,v_night_count,v_reservations_count
    from public.fo_schedule_entries where schedule_date=v_date;

    select exists(
      select 1 from public.fo_schedule_entries e join public.fo_schedule_staff s on s.id=e.staff_id
      where e.schedule_date=v_date and e.status='WORK'
        and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
        and lower(trim(s.staff_name))='saravanan'
    ) into v_saravanan_present;

    -- Count prior solo Saravanan Night dates inside the same Monday-Sunday week.
    -- A maximum of two such dates may release scarce manpower into daytime.
    select count(*) into v_solo_nights_week
    from (
      select e.schedule_date
      from public.fo_schedule_entries e
      join public.fo_schedule_staff s on s.id=e.staff_id
      where e.schedule_date>=date_trunc('week',v_date)::date
        and e.schedule_date<v_date and e.status='WORK'
        and upper(coalesce(e.shift_code_snapshot,'')) in ('NIGHT','COVER_NOON')
      group by e.schedule_date
      having count(*)=1 and bool_or(lower(trim(s.staff_name))='saravanan')
    ) solo_dates;

    -- When daytime coverage is short, use one of the two weekly solo-night
    -- allowances by moving Saravanan's auto-filled Night partner to the first
    -- unresolved daytime position. Manual entries are never moved.
    if v_saravanan_present and v_solo_nights_week<2 and v_night_count=2
      and (v_reservations_count<1 or v_morning_count<2 or v_afternoon_count<2) then
      v_rebalance_target:=case
        when v_reservations_count<1 then 'RESERVATIONS'
        when v_morning_count<2 then 'MORNING'
        else 'AFTERNOON'
      end;
      select e.id,e.staff_id into v_rebalance_entry_id,v_staff_id
      from public.fo_schedule_entries e
      join public.fo_schedule_staff s on s.id=e.staff_id
      where e.schedule_date=v_date and e.status='WORK' and e.auto_filled=true
        and upper(coalesce(e.shift_code_snapshot,''))='NIGHT'
        and lower(trim(s.staff_name))<>'saravanan'
      order by case when lower(trim(s.staff_name))='harish' then 1 else 0 end,
        s.sort_order,s.staff_name limit 1;
      if v_rebalance_entry_id is not null then
        select * into v_shift from public.fo_schedule_shifts
        where is_active and upper(shift_code)=v_rebalance_target limit 1;
        v_prev_code:=null; v_prev_date:=null; v_prev_start:=null; v_prev_end:=null; v_rest_hours:=null;
        select upper(coalesce(shift_code_snapshot,'')),schedule_date,scheduled_start,scheduled_end
          into v_prev_code,v_prev_date,v_prev_start,v_prev_end
        from public.fo_schedule_entries
        where staff_id=v_staff_id and schedule_date<v_date and status='WORK'
        order by schedule_date desc limit 1;
        v_rest_hours:=public.fo_schedule_rest_hours(v_prev_date,v_prev_start,v_prev_end,v_date,v_shift.start_time);
        v_is_exception:=coalesce(v_prev_code<>v_rebalance_target and v_rest_hours<16,false);
        update public.fo_schedule_entries set
          shift_id=v_shift.id,shift_name_snapshot=v_shift.shift_name,shift_code_snapshot=v_shift.shift_code,
          scheduled_start=v_shift.start_time,scheduled_end=v_shift.end_time,
          notes=concat_ws(' ',case when v_is_exception then 'Rest exception: '||v_prev_code||' to '||v_rebalance_target||' provides only '||trim(to_char(v_rest_hours,'FM999990.00'))||' hours before the next shift.' end,
            'Weekly solo-night optimisation: reassigned from Night to protect daytime coverage.'),
          compliance_exception=v_is_exception,updated_at=now()
        where id=v_rebalance_entry_id;
        v_night_count:=v_night_count-1;
        if v_rebalance_target='RESERVATIONS' then v_reservations_count:=v_reservations_count+1;
        elsif v_rebalance_target='MORNING' then v_morning_count:=v_morning_count+1;
        else v_afternoon_count:=v_afternoon_count+1; end if;
        if v_is_exception then v_exceptions:=v_exceptions+1; end if;
        v_saravanan_solo:=true; v_rebalance_entry_id:=null; v_staff_id:=null;
      end if;
    end if;

    while v_saravanan_present and v_afternoon_count<2 and v_morning_count>=2
      and not exists(
        select 1 from public.fo_schedule_entries cm
        where cm.schedule_date=v_date and cm.status='WORK'
          and upper(coalesce(cm.shift_code_snapshot,''))='COVER_MORNING'
      ) loop
      select e.staff_id into v_staff_id
      from public.fo_schedule_entries e
      join public.fo_schedule_staff s on s.id=e.staff_id
      where e.schedule_date=v_date and e.status='WORK' and e.auto_filled=true
        and upper(coalesce(e.shift_code_snapshot,''))='MORNING'
        and not s.is_manager
      order by case when lower(trim(s.staff_name))='harish' then 1 else 0 end,s.sort_order,s.staff_name
      limit 1;
      exit when v_staff_id is null;
      select * into v_shift from public.fo_schedule_shifts where is_active and upper(shift_code)='COVER_MORNING' limit 1;
      update public.fo_schedule_entries set
        shift_id=v_shift.id,shift_name_snapshot=v_shift.shift_name,shift_code_snapshot=v_shift.shift_code,
        scheduled_start=v_shift.start_time,scheduled_end=v_shift.end_time,
        notes=concat_ws(' ',nullif(notes,''),'OT bridge: covers Morning and Afternoon until 7:30 PM.'),updated_at=now()
      where staff_id=v_staff_id and schedule_date=v_date;
      v_afternoon_count:=v_afternoon_count+1; v_staff_id:=null;
    end loop;

    v_saravanan_solo:=v_saravanan_present and v_night_count=1 and v_solo_nights_week<2;

    while v_saravanan_present and v_night_count<2 and not v_saravanan_solo loop
      select e.staff_id into v_staff_id
      from public.fo_schedule_entries e
      join public.fo_schedule_staff s on s.id=e.staff_id
      where e.schedule_date=v_date and e.status='WORK' and e.auto_filled=true
        and upper(coalesce(e.shift_code_snapshot,''))='AFTERNOON'
        and not s.is_manager
        and (
          exists(
            select 1 from public.fo_schedule_entries n join public.fo_schedule_staff ns on ns.id=n.staff_id
            where n.schedule_date=v_date and n.status='WORK'
              and upper(coalesce(n.shift_code_snapshot,'')) in ('NIGHT','COVER_NOON') and ns.staff_gender='MALE'
          ) or s.staff_gender='MALE'
        )
        and (
          not exists(
            select 1 from public.fo_schedule_entries n join public.fo_schedule_staff ns on ns.id=n.staff_id
            where n.schedule_date=v_date and n.status='WORK'
              and upper(coalesce(n.shift_code_snapshot,'')) in ('NIGHT','COVER_NOON')
              and lower(trim(ns.staff_name))='jia'
          ) or lower(trim(s.staff_name)) in ('harish','zaim')
        )
      order by case when lower(trim(s.staff_name))='harish' then 1 else 0 end,s.sort_order,s.staff_name
      limit 1;
      exit when v_staff_id is null;
      select * into v_shift from public.fo_schedule_shifts where is_active and upper(shift_code)='COVER_NOON' limit 1;
      update public.fo_schedule_entries set
        shift_id=v_shift.id,shift_name_snapshot=v_shift.shift_name,shift_code_snapshot=v_shift.shift_code,
        scheduled_start=v_shift.start_time,scheduled_end=v_shift.end_time,
        notes=concat_ws(' ',nullif(notes,''),'OT bridge: covers late Afternoon and early Night.'),updated_at=now()
      where staff_id=v_staff_id and schedule_date=v_date;
      v_night_count:=v_night_count+1; v_staff_id:=null;
    end loop;

    select exists(
      select 1 from public.fo_schedule_entries e
      where e.schedule_date=v_date and e.status='WORK'
        and upper(coalesce(e.shift_code_snapshot,'')) in ('COVER_MORNING','COVER_NOON')
    ) into v_cover_mode;

    -- Walter normally remains on Manager hours. During the Saravanan-only
    -- staggered fallback he works Cover Manager (11 AM-7 PM), backing up Cover
    -- Morning after the regular Morning shift ends. Otherwise his last-resort
    -- order remains Reservations, Morning, Afternoon, then Night.
    select s.id into v_staff_id from public.fo_schedule_staff s
    where s.is_active and s.is_manager
      and not exists(select 1 from public.fo_schedule_entries e where e.staff_id=s.id and e.schedule_date=v_date)
    order by s.sort_order,s.staff_name limit 1;
    if v_staff_id is not null then
      v_manager_target:=case
        when v_reservations_count<1 then 'RESERVATIONS'
        when v_morning_count<2 then 'MORNING'
        when v_saravanan_present and v_cover_mode then 'COVER_MANAGER'
        when v_afternoon_count<2 then 'AFTERNOON'
        when v_night_count<2 and not v_saravanan_solo then 'NIGHT'
        else 'MANAGER'
      end;
      select * into v_shift from public.fo_schedule_shifts where is_active and upper(shift_code)=v_manager_target limit 1;
      insert into public.fo_schedule_entries(staff_id,schedule_date,status,shift_id,shift_name_snapshot,shift_code_snapshot,scheduled_start,scheduled_end,auto_filled,updated_by_user_id,updated_by_name)
      values(v_staff_id,v_date,'WORK',v_shift.id,v_shift.shift_name,v_shift.shift_code,v_shift.start_time,v_shift.end_time,true,(select auth.uid()),public.fo_schedule_actor_name())
      on conflict do nothing;
      get diagnostics v_current=row_count; v_inserted:=v_inserted+v_current;
      if v_current>0 then
        if v_manager_target='RESERVATIONS' then v_reservations_count:=v_reservations_count+1;
        elsif v_manager_target='MORNING' then v_morning_count:=v_morning_count+1;
        elsif v_manager_target in ('AFTERNOON','COVER_MANAGER') then v_afternoon_count:=v_afternoon_count+1;
        elsif v_manager_target='NIGHT' then v_night_count:=v_night_count+1; end if;
      end if;
    end if;

    v_saravanan_solo:=v_saravanan_present and v_night_count=1 and v_solo_nights_week<2;
    v_shortages:=v_shortages+greatest(0,2-v_morning_count)+greatest(0,2-v_afternoon_count)+greatest(0,1-v_reservations_count);
    if v_night_count<2 and not v_saravanan_solo then v_shortages:=v_shortages+(2-v_night_count); end if;
    select count(*) into v_night_male
    from public.fo_schedule_entries e join public.fo_schedule_staff s on s.id=e.staff_id
    where e.schedule_date=v_date and e.status='WORK'
      and upper(coalesce(e.shift_code_snapshot,'')) in ('NIGHT','COVER_NOON') and s.staff_gender='MALE';
    if v_night_count>0 and v_night_male=0 then v_shortages:=v_shortages+1; end if;

    -- Mid is surplus coverage only. Never create any Mid assignment until all
    -- standard targets, including two Night staff, are actually filled.
    select
      count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,'')) in ('MORNING','COVER_MORNING'))>=2
      and count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,'')) in ('AFTERNOON','COVER_MORNING','COVER_NOON','COVER_MANAGER'))>=2
      and count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,'')) in ('NIGHT','COVER_NOON'))>=2
      and count(*) filter (where status='WORK' and upper(coalesce(shift_code_snapshot,''))='RESERVATIONS')>=1
    into v_core_ready
    from public.fo_schedule_entries
    where schedule_date=v_date;

    if coalesce(v_core_ready,false) then
      select * into v_shift from public.fo_schedule_shifts where is_active and upper(shift_code)='MID' limit 1;
      insert into public.fo_schedule_entries(staff_id,schedule_date,status,shift_id,shift_name_snapshot,shift_code_snapshot,scheduled_start,scheduled_end,auto_filled,updated_by_user_id,updated_by_name)
      select s.id,v_date,'WORK',v_shift.id,v_shift.shift_name,v_shift.shift_code,v_shift.start_time,v_shift.end_time,true,(select auth.uid()),public.fo_schedule_actor_name()
      from public.fo_schedule_staff s
      where s.is_active and not s.is_manager
        and not exists(select 1 from public.fo_schedule_entries e where e.staff_id=s.id and e.schedule_date=v_date)
      order by case when lower(trim(s.staff_name))='harish' then 0 else 1 end,s.sort_order,s.staff_name on conflict do nothing;
      get diagnostics v_current=row_count; v_inserted:=v_inserted+v_current;
    end if;
  end loop;
  return jsonb_build_object('replaced',v_replaced,'inserted',v_inserted,'exceptions',v_exceptions,'shortages',v_shortages);
end $$;

-- Sri's normal assignment is Afternoon. She remains eligible for Night as a
-- hybrid, while Auto Fill avoids assigning her to Morning unless unavoidable.
update public.fo_schedule_staff
set preferred_shift='AFTERNOON', updated_at=now()
where lower(trim(staff_name))='sri';

insert into public.fo_schedule_shifts(shift_name,shift_code,start_time,end_time,color,is_active,created_by_user_id,created_by_name)
select x.name,x.code,x.start_at,x.end_at,x.color,true,coalesce((select user_id from public.user_profiles where upper(role)='SUPERUSER' limit 1),'00000000-0000-0000-0000-000000000000'::uuid),'System'
from (values
  ('Morning','MORNING','07:00'::time,'15:30'::time,'BLUE'),
  ('Afternoon','AFTERNOON','14:30'::time,'23:00'::time,'PURPLE'),
  ('Night','NIGHT','22:45'::time,'07:15'::time,'SLATE'),
  ('Reservations','RESERVATIONS','09:00'::time,'17:30'::time,'TEAL'),
  ('Cover Morning','COVER_MORNING','09:00'::time,'19:30'::time,'AMBER'),
