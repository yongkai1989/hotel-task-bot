-- Run this only after the matching application update has been deployed.
-- It moves dashboard task synchronization away from Postgres Changes, which
-- removes the high-volume realtime.list_changes workload from public.tasks.

drop policy if exists "dashboard users receive task broadcasts"
on realtime.messages;

create policy "dashboard users receive task broadcasts"
on realtime.messages
for select
to authenticated
using (realtime.topic() = 'dashboard-task-sync');

do $$
begin
  if exists (
    select 1
    from pg_publication_tables
    where pubname = 'supabase_realtime'
      and schemaname = 'public'
      and tablename = 'tasks'
  ) then
    execute 'alter publication supabase_realtime drop table public.tasks';
  end if;
end
$$;
