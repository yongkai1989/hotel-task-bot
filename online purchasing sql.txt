-- Online Purchasing
-- Run this entire file once in the Supabase SQL Editor.

begin;

alter table public.user_profiles
  add column if not exists can_access_online_purchasing boolean not null default false;

create table if not exists public.online_purchasing_hotels (
  code text primary key,
  name text not null,
  legal_entity text not null,
  opening_float numeric(12,2) not null default 1000 check (opening_float >= 0),
  sort_order integer not null default 0,
  active boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

insert into public.online_purchasing_hotels (code, name, legal_entity, opening_float, sort_order)
values
  ('CROWN', 'Hallmark Crown Hotel', 'Hotel Hallmark Leisure Sdn Bhd', 1000, 1),
  ('LEISURE', 'Hallmark Leisure Hotel', 'Hotel Hallmark Leisure Sdn Bhd', 1000, 2),
  ('EXPRESS', 'Hallmark Express Hotel', 'Hotel Hallmark Express Sdn Bhd', 1000, 3),
  ('VIEW', 'Hallmark View Hotel', 'Hallmark View Hotel Sdn Bhd', 1000, 4)
on conflict (code) do update set
  name = excluded.name,
  legal_entity = excluded.legal_entity,
  sort_order = excluded.sort_order,
  updated_at = now();

create table if not exists public.online_purchasing_user_access (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  hotel_code text not null references public.online_purchasing_hotels(code),
  access_role text not null check (access_role in ('PURCHASER', 'HOD')),
  created_by uuid references auth.users(id),
  created_at timestamptz not null default now(),
  unique (user_id, hotel_code, access_role)
);

create table if not exists public.online_purchase_orders (
  id uuid primary key default gen_random_uuid(),
  purchase_number bigint generated by default as identity unique,
  hotel_code text not null references public.online_purchasing_hotels(code),
  shopee_url text not null check (shopee_url ~* '^https?://'),
  status text not null default 'AWAITING_PURCHASE' check (status in (
    'AWAITING_PURCHASE',
    'PURCHASED_IN_TRANSIT',
    'ARRIVED_INVOICE_PENDING',
    'PENDING_REIMBURSEMENT',
    'COMPLETE_CLAIMED',
    'REFUND_PENDING',
    'REFUND_COMPLETED',
    'CANCELLED'
  )),
  purchase_amount numeric(12,2) check (purchase_amount is null or purchase_amount > 0),
  purchased_at timestamptz,
  purchased_by uuid references auth.users(id),
  arrived_at timestamptz,
  arrived_by uuid references auth.users(id),
  invoice_submitted_at timestamptz,
  invoice_submitted_by uuid references auth.users(id),
  reimbursed_at timestamptz,
  reimbursed_by uuid references auth.users(id),
  reimbursement_amount numeric(12,2) check (reimbursement_amount is null or reimbursement_amount > 0),
  refund_issue_type text check (refund_issue_type is null or refund_issue_type in ('LOST', 'DEFECTIVE', 'OTHER')),
  refund_remark text,
  refund_requested_at timestamptz,
  refund_requested_by uuid references auth.users(id),
  refund_completed_at timestamptz,
  refund_completed_by uuid references auth.users(id),
  refund_amount numeric(12,2) check (refund_amount is null or refund_amount > 0),
  cancelled_at timestamptz,
  cancelled_by uuid references auth.users(id),
  created_by uuid not null references auth.users(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.online_purchase_documents (
  id uuid primary key default gen_random_uuid(),
  order_id uuid not null references public.online_purchase_orders(id) on delete restrict,
  document_type text not null check (document_type in ('E_INVOICE', 'PHOTO', 'REFUND_PHOTO')),
  storage_path text not null unique,
  file_name text not null,
  mime_type text not null,
  file_size bigint not null check (file_size > 0),
  uploaded_by uuid not null references auth.users(id),
  created_at timestamptz not null default now()
);

create table if not exists public.online_purchase_ledger_entries (
  id uuid primary key default gen_random_uuid(),
  hotel_code text not null references public.online_purchasing_hotels(code),
  order_id uuid not null references public.online_purchase_orders(id) on delete restrict,
  entry_type text not null check (entry_type in ('PURCHASE', 'REIMBURSEMENT', 'REFUND')),
  amount_delta numeric(12,2) not null check (amount_delta <> 0),
  actor_user_id uuid not null references auth.users(id),
  created_at timestamptz not null default now(),
  unique (order_id, entry_type)
);

create table if not exists public.online_purchase_audit (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references public.online_purchase_orders(id) on delete restrict,
  hotel_code text references public.online_purchasing_hotels(code),
  action text not null,
  previous_status text,
  new_status text,
  details jsonb not null default '{}'::jsonb,
  actor_user_id uuid not null references auth.users(id),
  actor_name text not null,
  actor_email text not null,
  created_at timestamptz not null default now()
);

create index if not exists online_purchase_orders_hotel_status_idx
  on public.online_purchase_orders (hotel_code, status, created_at desc);
create index if not exists online_purchase_orders_created_idx
  on public.online_purchase_orders (created_at desc);
create index if not exists online_purchase_documents_order_idx
  on public.online_purchase_documents (order_id, created_at);
create index if not exists online_purchase_ledger_hotel_idx
  on public.online_purchase_ledger_entries (hotel_code, created_at);
create index if not exists online_purchase_audit_order_idx
  on public.online_purchase_audit (order_id, created_at);
create index if not exists online_purchasing_access_user_idx
  on public.online_purchasing_user_access (user_id, hotel_code);

alter table public.online_purchasing_hotels enable row level security;
alter table public.online_purchasing_user_access enable row level security;
alter table public.online_purchase_orders enable row level security;
alter table public.online_purchase_documents enable row level security;
alter table public.online_purchase_ledger_entries enable row level security;
alter table public.online_purchase_audit enable row level security;

revoke all on table public.online_purchasing_hotels from anon, authenticated;
revoke all on table public.online_purchasing_user_access from anon, authenticated;
revoke all on table public.online_purchase_orders from anon, authenticated;
revoke all on table public.online_purchase_documents from anon, authenticated;
revoke all on table public.online_purchase_ledger_entries from anon, authenticated;
revoke all on table public.online_purchase_audit from anon, authenticated;
grant all on table public.online_purchasing_hotels to service_role;
grant all on table public.online_purchasing_user_access to service_role;
grant all on table public.online_purchase_orders to service_role;
grant all on table public.online_purchase_documents to service_role;
grant all on table public.online_purchase_ledger_entries to service_role;
grant all on table public.online_purchase_audit to service_role;
grant usage, select on all sequences in schema public to service_role;

create or replace function public.online_purchasing_apply_action(
  p_actor_user_id uuid,
  p_action text,
  p_order_id uuid default null,
  p_payload jsonb default '{}'::jsonb
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_actor public.user_profiles%rowtype;
  v_order public.online_purchase_orders%rowtype;
  v_hotel_code text;
  v_previous_status text;
  v_amount numeric(12,2);
  v_issue text;
  v_remark text;
  v_has_assignment boolean;
begin
  select * into v_actor
  from public.user_profiles
  where user_id = p_actor_user_id;

  if not found then
    raise exception 'User profile not found';
  end if;

  if v_actor.role <> 'SUPERUSER' and not coalesce(v_actor.can_access_online_purchasing, false) then
    raise exception 'Online Purchasing access denied';
  end if;

  if p_action = 'CREATE_ORDER' then
    if v_actor.role <> 'SUPERUSER' then raise exception 'Only Superuser can create orders'; end if;
    v_hotel_code := upper(trim(p_payload->>'hotel_code'));
    if not exists (select 1 from public.online_purchasing_hotels where code = v_hotel_code and active) then
      raise exception 'Invalid hotel';
    end if;
    if coalesce(trim(p_payload->>'shopee_url'), '') !~* '^https?://' then
      raise exception 'Enter a valid Shopee link';
    end if;

    insert into public.online_purchase_orders (hotel_code, shopee_url, created_by)
    values (v_hotel_code, trim(p_payload->>'shopee_url'), p_actor_user_id)
    returning * into v_order;

  else
    select * into v_order from public.online_purchase_orders where id = p_order_id for update;
    if not found then raise exception 'Order not found'; end if;
    v_previous_status := v_order.status;

    if p_action in ('SAVE_PURCHASE','MARK_ARRIVED','SUBMIT_DOCUMENTS','START_REFUND','COMPLETE_REFUND') then
      select exists (
        select 1 from public.online_purchasing_user_access
        where user_id = p_actor_user_id
          and hotel_code = v_order.hotel_code
          and access_role = 'PURCHASER'
      ) into v_has_assignment;
      if not v_has_assignment then raise exception 'Purchaser access for this hotel is required'; end if;
    end if;

    if p_action = 'SAVE_PURCHASE' then
      if v_order.status <> 'AWAITING_PURCHASE' then raise exception 'This order is no longer awaiting purchase'; end if;
      v_amount := nullif(p_payload->>'amount','')::numeric;
      if v_amount is null or v_amount <= 0 then raise exception 'Enter the total purchased amount'; end if;
      update public.online_purchase_orders set
        status = 'PURCHASED_IN_TRANSIT', purchase_amount = v_amount,
        purchased_at = now(), purchased_by = p_actor_user_id, updated_at = now()
      where id = v_order.id returning * into v_order;
      insert into public.online_purchase_ledger_entries
        (hotel_code, order_id, entry_type, amount_delta, actor_user_id)
      values (v_order.hotel_code, v_order.id, 'PURCHASE', -v_amount, p_actor_user_id);

    elsif p_action = 'MARK_ARRIVED' then
      if v_order.status <> 'PURCHASED_IN_TRANSIT' then raise exception 'Only an in-transit order can be marked arrived'; end if;
      update public.online_purchase_orders set
        status = 'ARRIVED_INVOICE_PENDING', arrived_at = now(), arrived_by = p_actor_user_id, updated_at = now()
      where id = v_order.id returning * into v_order;

    elsif p_action = 'SUBMIT_DOCUMENTS' then
      if v_order.status <> 'ARRIVED_INVOICE_PENDING' then raise exception 'This order is not awaiting an e-Invoice'; end if;
      if not exists (
        select 1 from public.online_purchase_documents
        where order_id = v_order.id and document_type = 'E_INVOICE'
      ) then raise exception 'Upload the e-Invoice PDF before submitting'; end if;
      update public.online_purchase_orders set
        status = 'PENDING_REIMBURSEMENT', invoice_submitted_at = now(),
        invoice_submitted_by = p_actor_user_id, updated_at = now()
      where id = v_order.id returning * into v_order;

    elsif p_action = 'COMPLETE_CLAIM' then
      select exists (
        select 1 from public.online_purchasing_user_access
        where user_id = p_actor_user_id
          and hotel_code = v_order.hotel_code
          and access_role = 'HOD'
      ) into v_has_assignment;
      if not v_has_assignment then raise exception 'HOD access for this hotel is required'; end if;
      if v_order.status <> 'PENDING_REIMBURSEMENT' then raise exception 'This order is not pending reimbursement'; end if;
      v_amount := v_order.purchase_amount;
      update public.online_purchase_orders set
        status = 'COMPLETE_CLAIMED', reimbursement_amount = v_amount,
        reimbursed_at = now(), reimbursed_by = p_actor_user_id, updated_at = now()
      where id = v_order.id returning * into v_order;
      insert into public.online_purchase_ledger_entries
        (hotel_code, order_id, entry_type, amount_delta, actor_user_id)
      values (v_order.hotel_code, v_order.id, 'REIMBURSEMENT', v_amount, p_actor_user_id);

    elsif p_action = 'START_REFUND' then
      if v_order.status not in ('PURCHASED_IN_TRANSIT','ARRIVED_INVOICE_PENDING','PENDING_REIMBURSEMENT') then
        raise exception 'This order cannot be moved to refund follow-up';
      end if;
      v_issue := upper(trim(p_payload->>'issue_type'));
      v_remark := nullif(trim(p_payload->>'remark'), '');
      if v_issue not in ('LOST','DEFECTIVE','OTHER') then raise exception 'Select Lost, Defective, or Other'; end if;
      update public.online_purchase_orders set
        status = 'REFUND_PENDING', refund_issue_type = v_issue, refund_remark = v_remark,
        refund_requested_at = now(), refund_requested_by = p_actor_user_id, updated_at = now()
      where id = v_order.id returning * into v_order;

    elsif p_action = 'COMPLETE_REFUND' then
      if v_order.status <> 'REFUND_PENDING' then raise exception 'This order is not pending refund'; end if;
      v_amount := coalesce(nullif(p_payload->>'amount','')::numeric, v_order.purchase_amount);
      if v_amount <= 0 or v_amount > v_order.purchase_amount then
        raise exception 'Refund amount must be more than zero and not exceed the purchase amount';
      end if;
      update public.online_purchase_orders set
        status = 'REFUND_COMPLETED', refund_amount = v_amount,
        refund_completed_at = now(), refund_completed_by = p_actor_user_id, updated_at = now()
      where id = v_order.id returning * into v_order;
      insert into public.online_purchase_ledger_entries
        (hotel_code, order_id, entry_type, amount_delta, actor_user_id)
      values (v_order.hotel_code, v_order.id, 'REFUND', v_amount, p_actor_user_id);

    elsif p_action = 'CANCEL_ORDER' then
      if v_actor.role <> 'SUPERUSER' then raise exception 'Only Superuser can cancel orders'; end if;
      if v_order.status <> 'AWAITING_PURCHASE' then raise exception 'Only an unpurchased order can be cancelled'; end if;
      update public.online_purchase_orders set
        status = 'CANCELLED', cancelled_at = now(), cancelled_by = p_actor_user_id, updated_at = now()
      where id = v_order.id returning * into v_order;
    else
      raise exception 'Unsupported action';
    end if;
  end if;

  insert into public.online_purchase_audit (
    order_id, hotel_code, action, previous_status, new_status, details,
    actor_user_id, actor_name, actor_email
  ) values (
    v_order.id, v_order.hotel_code, p_action, v_previous_status, v_order.status,
    coalesce(p_payload, '{}'::jsonb), p_actor_user_id,
    coalesce(v_actor.name, v_actor.email, 'User'), coalesce(v_actor.email, '')
  );

  return jsonb_build_object('id', v_order.id, 'status', v_order.status);
end;
$$;

revoke all on function public.online_purchasing_apply_action(uuid,text,uuid,jsonb) from public, anon, authenticated;
grant execute on function public.online_purchasing_apply_action(uuid,text,uuid,jsonb) to service_role;

commit;
