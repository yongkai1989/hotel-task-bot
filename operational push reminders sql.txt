-- Operational Web Push reminders
-- Safe to run more than once in the Supabase SQL Editor.

create table if not exists public.daily_operational_notification_runs (
  notification_date date not null,
  notification_type text not null check (
    notification_type in ('CHAMBERMAID_5PM', 'PREVENTIVE_MAINTENANCE_9AM')
  ),
  status text not null check (
    status in ('SENDING', 'SENT', 'NOT_NEEDED', 'FAILED')
  ),
  attempted_at timestamptz not null default now(),
  sent_at timestamptz,
  finding_count integer not null default 0,
  attempted_count integer not null default 0,
  delivered_count integer not null default 0,
  telegram_message_id bigint,
  details jsonb,
  error_text text,
  primary key (notification_date, notification_type)
);

alter table public.daily_operational_notification_runs enable row level security;
revoke all on table public.daily_operational_notification_runs from public, anon, authenticated;
grant all on table public.daily_operational_notification_runs to service_role;

comment on table public.daily_operational_notification_runs is
  'Idempotency and delivery audit for the 5 PM chambermaid and 9 AM overdue PM reminders.';

-- Keep the database-side timed-popup recipient trigger aligned with the app.
-- The additional Maintenance account is matched by its saved profile name so its
-- email address can remain whatever was configured in Admin Settings.
create or replace function public.create_task_alert_recipients()
returns trigger
language plpgsql
set search_path = public
as $$
begin
  if new.status = 'OPEN' and (new.customer_waiting = true or new.urgent = true) then
    insert into public.task_alert_recipients (
      task_id,
      alert_cycle,
      user_id,
      user_name,
      user_email
    )
    select
      new.id,
      greatest(coalesce(new.alert_cycle, 1), 1),
      profile.user_id,
      coalesce(nullif(trim(profile.name), ''), profile.email),
      lower(trim(profile.email))
    from public.user_profiles profile
    where (
      lower(trim(profile.email)) = any (
        case upper(trim(new.department))
          when 'HK' then array[
            'hksup1@hotelhallmark.com',
            'hksup2@hotelhallmark.com',
            'hksup3@hotelhallmark.com',
            'manager@hotelhallmark.com'
          ]::text[]
          when 'MT' then array[
            'mtsup1@hotelhallmark.com',
            'mtsup2@hotelhallmark.com'
          ]::text[]
          else array[]::text[]
        end
      )
      or (
        upper(trim(new.department)) = 'MT'
        and lower(trim(profile.name)) = 'maintenance'
      )
    )
    on conflict (task_id, alert_cycle, user_id) do nothing;
  end if;

  return new;
end;
$$;

revoke all on function public.create_task_alert_recipients() from public, anon, authenticated;
