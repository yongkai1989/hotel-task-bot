-- Commission Checker: superuser-managed username and passcode access
-- Run this complete file once in the Supabase SQL Editor.

create table if not exists public.commission_checker_access (
  id uuid primary key default gen_random_uuid(),
  email text not null,
  label text not null default '',
  is_active boolean not null default true,
  session_version integer not null default 1 check (session_version > 0),
  created_by uuid null references auth.users(id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  last_login_at timestamptz null,
  passcode_hash text null,
  passcode_generated_at timestamptz null,
  passcode_expires_at timestamptz null,
  passcode_never_expires boolean not null default false,
  passcode_generated_by uuid null references auth.users(id) on delete set null,
  constraint commission_checker_access_email_normalized
    check (email = lower(trim(email)))
);

-- These statements safely upgrade installations that used the previous email sign-in flow.
alter table public.commission_checker_access
  add column if not exists passcode_hash text null,
  add column if not exists passcode_generated_at timestamptz null,
  add column if not exists passcode_expires_at timestamptz null,
  add column if not exists passcode_never_expires boolean not null default false,
  add column if not exists passcode_generated_by uuid null;

do $$
begin
  if not exists (
    select 1
    from pg_constraint
    where conname = 'commission_checker_access_passcode_generated_by_fkey'
      and conrelid = 'public.commission_checker_access'::regclass
  ) then
    alter table public.commission_checker_access
      add constraint commission_checker_access_passcode_generated_by_fkey
      foreign key (passcode_generated_by) references auth.users(id) on delete set null;
  end if;
end
$$;

do $$
begin
  if not exists (
    select 1
    from pg_constraint
    where conname = 'commission_checker_access_permanent_email_check'
      and conrelid = 'public.commission_checker_access'::regclass
  ) then
    alter table public.commission_checker_access
      add constraint commission_checker_access_permanent_email_check
      check (
        passcode_never_expires = false
        or email = 'ryan.tan@hotelhallmark.com'
      );
  end if;
end
$$;

create unique index if not exists commission_checker_access_email_unique
  on public.commission_checker_access (email);

create index if not exists commission_checker_access_active_idx
  on public.commission_checker_access (is_active, email);

create table if not exists public.commission_checker_auth_attempts (
  id bigint generated by default as identity primary key,
  email text not null,
  action text not null check (action in ('request', 'verify', 'login')),
  successful boolean not null default false,
  ip_address text null,
  created_at timestamptz not null default now()
);

-- Preserve existing OTP audit rows while allowing the new passcode login action.
alter table public.commission_checker_auth_attempts
  drop constraint if exists commission_checker_auth_attempts_action_check;

alter table public.commission_checker_auth_attempts
  add constraint commission_checker_auth_attempts_action_check
  check (action in ('request', 'verify', 'login'));

create index if not exists commission_checker_auth_attempts_limit_idx
  on public.commission_checker_auth_attempts (email, action, created_at desc);

alter table public.commission_checker_access enable row level security;
alter table public.commission_checker_auth_attempts enable row level security;

-- No browser policies are intentionally created. Only the server-side service role
-- can read or change usernames, passcode hashes, and authentication attempts.
revoke all on table public.commission_checker_access from anon, authenticated;
revoke all on table public.commission_checker_auth_attempts from anon, authenticated;
revoke all on sequence public.commission_checker_auth_attempts_id_seq from anon, authenticated;

-- Keep authentication attempt records for 30 days. This function is not scheduled;
-- it runs opportunistically whenever an authentication attempt is inserted.
create or replace function public.cleanup_old_commission_checker_auth_attempts()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if random() < 0.05 then
    delete from public.commission_checker_auth_attempts
    where created_at < now() - interval '30 days';
  end if;
  return new;
end;
$$;

revoke all on function public.cleanup_old_commission_checker_auth_attempts() from public, anon, authenticated;
grant execute on function public.cleanup_old_commission_checker_auth_attempts() to service_role;

drop trigger if exists commission_checker_auth_attempts_cleanup
  on public.commission_checker_auth_attempts;

create trigger commission_checker_auth_attempts_cleanup
after insert on public.commission_checker_auth_attempts
for each row execute function public.cleanup_old_commission_checker_auth_attempts();

grant usage on schema public to service_role;
grant all on table public.commission_checker_access to service_role;
grant all on table public.commission_checker_auth_attempts to service_role;
grant usage, select on sequence public.commission_checker_auth_attempts_id_seq to service_role;

notify pgrst, 'reload schema';
